cmake_minimum_required(VERSION 3.16)
project(iso8583_course_environment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Boost REQUIRED)
add_executable(course_check src/main.cpp)
target_include_directories(course_check PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(course_check PRIVATE Threads::Threads OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3)
