cmake_minimum_required(VERSION 3.16)
project(iso8583_day03_gateway LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(iso8583_parser STATIC src/iso8583_parser.cpp)
target_include_directories(iso8583_parser PUBLIC include)
target_compile_options(iso8583_parser PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
find_package(Threads REQUIRED)
# Boost.System is header-only for modern Boost; avoid mandatory binary package.
add_executable(gateway src/main.cpp src/gateway.cpp src/session.cpp src/logger.cpp)
target_include_directories(gateway PRIVATE include)
target_compile_options(gateway PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_compile_definitions(gateway PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(gateway PRIVATE iso8583_parser Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_parser tests/test_parser.cpp)
  target_link_libraries(test_parser PRIVATE iso8583_parser)
  add_test(NAME parser_complete COMMAND test_parser)
endif()

# Day 4 infrastructure skeleton: intentionally incomplete.
find_package(SQLite3 REQUIRED)
add_library(database STATIC src/database.cpp)
target_compile_features(database PUBLIC cxx_std_17)
target_include_directories(database PUBLIC include)
target_link_libraries(database PUBLIC SQLite::SQLite3)
target_link_libraries(gateway PRIVATE database)
